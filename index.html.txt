        const tabsContainer = document.getElementById('tabs-container');

                }
            });

            tabContents.forEach(content => {
                content.classList.remove('active');
                if (content.id === tabId) {
                    content.classList.add('active');
                }
            });

        function copyCode(button) {
            const codeBlock = button.nextElementSibling;
            const text = codeBlock.innerText;

            const textArea = document.createElement('textarea');
            textArea.value = text;
            document.body.appendChild(textArea);
            textArea.select();
            try {
                document.execCommand('copy');
                button.innerText = 'Copiado!';
                setTimeout(() => {
                    button.innerText = 'Copiar';
                }, 2000);
            } catch (err) {
                console.error('Failed to copy text: ', err);
            }
            document.body.removeChild(textArea);
        }

        const originalLuaScript = `
-- ================== CONFIGURACIÓN ==================
-- Configuración de Sonido
local FADE_DURATION = 1.0  -- Duración del fundido de sonido
local TARGET_VOLUME = 0.6  -- Volumen máximo del sonido

-- Configuración de Cámara
local USE_FOV_EFFECT = true
local ZONE_FOV = 86
local FOV_FADE_DURATION = 1.75
-- =================================================

-- Servicios
local TweenService = game:GetService("TweenService")
local RunService = game:GetService("RunService")
local Players = game:GetService("Players")
local Workspace = game:GetService("Workspace")
local CollectionService = game:GetService("CollectionService")

local CameraEffects = require(script.CameraEffects)

local localPlayer = Players.LocalPlayer
local character = localPlayer.Character or localPlayer.CharacterAdded:Wait()
local humanoidRootPart = character:WaitForChild("HumanoidRootPart")

local soundPartsFolder = Workspace:WaitForChild("soundParts")

local masterSound = Instance.new("Sound")
masterSound.Name = "MasterAmbientSound"
masterSound.Looped = true
masterSound.Volume = 0
masterSound.Parent = script

local currentActiveZone, activeTween = nil, nil
local isTransitioning = false
local soundPlaybackPositions = {}

local overlapParams = OverlapParams.new()
overlapParams.FilterType = Enum.RaycastFilterType.Include
overlapParams.FilterDescendantsInstances = {soundPartsFolder}

local function onTweenCompleted()
	isTransitioning = false
	activeTween = nil
end

local function fadeIn()
	if activeTween then activeTween:Cancel() end
	masterSound:Play()
	activeTween = TweenService:Create(masterSound, TweenInfo.new(FADE_DURATION, Enum.EasingStyle.Linear), {Volume = TARGET_VOLUME})
	activeTween.Completed:Once(onTweenCompleted)
	activeTween:Play()
end

local function fadeOut(onCompleteCallback)
	if activeTween then activeTween:Cancel() end
	activeTween = TweenService:Create(masterSound, TweenInfo.new(FADE_DURATION, Enum.EasingStyle.Linear), {Volume = 0})
	activeTween.Completed:Once(function()
		if masterSound.Volume == 0 then masterSound:Pause() end
		activeTween = nil
		if onCompleteCallback then onCompleteCallback() else isTransitioning = false end
	end)
	activeTween:Play()
end

local function handleFOV(oldZone, newZone)
	if not USE_FOV_EFFECT then return end
	local oldTag = oldZone and CollectionService:HasTag(oldZone, "camera")
	local newTag = newZone and CollectionService:HasTag(newZone, "camera")
	if newTag and not oldTag then
		CameraEffects.setFOV(ZONE_FOV, FOV_FADE_DURATION)
	elseif not newTag and oldTag then
		CameraEffects.resetFOV(FOV_FADE_DURATION)
	end
end

local function onHeartbeat()
	if not humanoidRootPart or not humanoidRootPart.Parent then
		character = localPlayer.Character
		if not character then return end
		humanoidRootPart = character:WaitForChild("HumanoidRootPart")
		return
	end
	if isTransitioning then return end

	local touchingZones = Workspace:GetPartsInPart(humanoidRootPart, overlapParams)
	local newActiveZone = touchingZones[1]
	if newActiveZone == currentActiveZone then return end

	local oldSourceSound = currentActiveZone and currentActiveZone:FindFirstChildOfClass("Sound")
	local newSourceSound = newActiveZone and newActiveZone:FindFirstChildOfClass("Sound")
	local oldSoundId = oldSourceSound and oldSourceSound.SoundId or ""
	local newSoundId = newSourceSound and newSourceSound.SoundId or ""

	if newSoundId ~= "" and newSoundId == oldSoundId then
		handleFOV(currentActiveZone, newActiveZone)
		currentActiveZone = newActiveZone
		return
	end

	if currentActiveZone and masterSound.IsPlaying and oldSoundId ~= "" then
		soundPlaybackPositions[oldSoundId] = masterSound.TimePosition
	end

	handleFOV(currentActiveZone, newActiveZone)
	isTransitioning = true
	currentActiveZone = newActiveZone

	local function playNewSound()
		if newSoundId ~= "" then
			masterSound.SoundId = newSoundId
			masterSound.TimePosition = soundPlaybackPositions[newSoundId] or 0
			fadeIn()
		else
			masterSound.SoundId = ""
			isTransitioning = false
		end
	end

	if masterSound.IsPlaying then fadeOut(playNewSound) else playNewSound() end
end

RunService.Heartbeat:Connect(onHeartbeat)

        async function callGeminiApi() {
            loader.style.display = 'block';
            aiResponseContainer.style.display = 'none';
            aiResponseContainer.textContent = '';
            generateBtn.disabled = true;

            let chatHistory = [{ role: "user", parts: [{ text: prompt }] }];
            const payload = { contents: chatHistory };
            const apiKey = "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;
            
            let attempts = 0;
            const maxAttempts = 5;
            let delay = 1000;

            while (attempts < maxAttempts) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }

                    const result = await response.json();
                    
                    if (result.candidates && result.candidates.length > 0 &&
                        result.candidates[0].content && result.candidates[0].content.parts &&
                        result.candidates[0].content.parts.length > 0) {
                        const text = result.candidates[0].content.parts[0].text;
                        aiResponseContainer.textContent = text;
                        aiResponseContainer.style.display = 'block';
                    } else {
                         aiResponseContainer.textContent = 'No se pudo obtener una respuesta válida de la IA. La respuesta podría estar vacía o tener un formato inesperado.';
                         aiResponseContainer.style.display = 'block';
                    }
                    break; 
                } catch (error) {
                    attempts++;
                    if (attempts >= maxAttempts) {
                        aiResponseContainer.textContent = 'Error al contactar la API de IA después de varios intentos. Por favor, inténtalo de nuevo más tarde.';
                        aiResponseContainer.style.display = 'block';
                    } else {
                        await new Promise(res => setTimeout(res, delay));
                        delay *= 2;
                    }
                } finally {
                    loader.style.display = 'none';
                    generateBtn.disabled = false;
                }
            }
        }

        generateBtn.addEventListener('click', callGeminiApi);
